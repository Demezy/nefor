# MAG in Five Minutes

```lisp
;; Two semicolons start a comment. MAG programs are data expressions written
;; in prefix form: the function comes first, followed by its arguments.
;;
;;   (function argument-1 argument-2)

;; ------------------------------------------------------------
;; 1. Types and values
;; ------------------------------------------------------------

;; A nominal record type gives a fixed shape a reusable name.
(type Score {:label String :accepted Bool})

;; Decision is a sum type: every Decision is either Accepted or Rejected.
(type Accepted {:score Score})
(type Rejected {:score Score :reason String})
(type Decision (| Accepted Rejected))

;; Box has one type parameter named T.
(type Box [T] {:value T})

(let unit nil)
(let enabled true)
(let attempts 3)
(let ratio 3.5)
(let language "MAG")
(let numbers [1 2 3])
(let compiler {:name "mag" :enabled true :attempts 3})
(let counters (as (Map String Int) {:accepted 2 :rejected 1}))

;; Bindings use kebab-case names by convention. A source file and every
;; function body form a block: all `let` bindings in that block are peers, and
;; the final expression is the block's result. `?` has no special syntax; it
;; is simply useful in predicate names such as `accepted?`.

;; Record and map access produces ordinary values. Here compiler-name is
;; "mag", updated-compiler has :attempts 4, and accepted-count is 2.
(let compiler-name (get compiler "name"))
(let updated-compiler (assoc compiler "attempts" 4))
(let accepted-count (get counters "accepted"))

;; ------------------------------------------------------------
;; 2. Collections derive new immutable collections
;; ------------------------------------------------------------

;; first-number is 1 and number-count is 3. Removing position 1 produces
;; [1 3]; the original numbers remain [1 2 3]. An invalid position makes
;; compilation fail during eager evaluation.
(let first-number (first numbers))
(let number-count (count numbers))
(let without-middle (remove-at numbers 1))
(let joined (concat numbers [4 5]))
(let field-names (keys compiler))

;; ------------------------------------------------------------
;; 3. Typed values, control flow, and functions
;; ------------------------------------------------------------

(let score (as Score {:label "review" :accepted true}))
(let decision (if (get score "accepted")
        (as Decision (as Accepted {:score score}))
        (as Decision
          (as Rejected {:score score :reason "not accepted"}))))
(let boxed-score (as (Box Score) {:value score}))

;; A function is an ordinary value. Its body is a block whose final expression
;; is returned, so label-score maps Score to String.
(let label-score (fn [[value Score]] -> String
        (get value "label")))
(let label (label-score score))

;; Generic functions declare type parameters before value parameters. Calls
;; infer T: boxed-string is Box String and boxed-number is Box Int.
(let box (fn [T] [[value T]] -> (Box T)
        (as (Box T) {:value value})))
(let boxed-string (box "typed"))
(let boxed-number (box 42))
(let outcome (if enabled "enabled" "disabled"))
(let rendered (map (fn [[value Int]] -> String (str value)) numbers))
(let indexed (indexed-map
        (fn [[index Int] [value Int]] -> {:index Int :value Int}
          {:index index :value value})
        numbers))
(let only-two (filter (fn [[value Int]] -> Bool (= value 2)) numbers))
(let repeated (flat-map (fn [[value Int]] -> (List Int) [value value]) numbers))
(let copied (fold
        (fn [[out (List Int)] [value Int]] -> (List Int)
          (concat out [value]))
        (as (List Int) [])
        numbers))
(let ordered (sort-by (fn [[value Int]] -> String (str value)) [3 1 2]))

;; ------------------------------------------------------------
;; 4. Types can enter data through checked evidence
;; ------------------------------------------------------------

(let score-tag (type-tag Score))
(let score-descriptor (type-evidence score-tag))
(let score-identity (type-id score-descriptor))
(let score-schema (type-schema score-tag))

;; ------------------------------------------------------------
;; 5. The artifact
;; ------------------------------------------------------------

;; The artifact expression is the entry point's result, analogous to the value
;; returned from `main` in many languages. It can contain any serializable
;; value. The compiler places exactly this value in CompilationResult.artifact;
;; MAG adds no application fields to it.
(artifact {:unit unit
     :language language
     :compiler compiler
     :updated_compiler updated-compiler
     :decision decision
     :boxed_score boxed-score
     :boxed_string boxed-string
     :boxed_number boxed-number
     :outcome outcome
     :rendered rendered
     :indexed indexed
     :filtered only-two
     :repeated repeated
     :copied copied
     :ordered ordered
     :score_type score-descriptor
     :score_type_id score-identity
     :score_schema score-schema})

;; Modules use literal `(require "module.name")` forms. Text and JSON files are
;; immutable source inputs:
;;
;;   (require "modules.labels")
;;   (let prompt (read "prompts/reviewer.md"))
;;   (let manifest (read-json "config/toolsets.json"))
```
